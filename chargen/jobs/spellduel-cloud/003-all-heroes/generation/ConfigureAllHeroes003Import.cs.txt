 [MenuItem("SpellDuel/CharKit/Configure job 003 hero imports")]
 public static void Execute(){foreach(var id in Ids){string p="Assets/SpellDuel/Resources/CharKit/"+id+"_v1/"+id+".fbx";if(AssetImporter.GetAtPath(p)!=null)Configure(p,id);}}
 public static void Configure(string path,string id){
  if(!Ids.Contains(id))throw new ArgumentException("Unknown hero id");
  var model=AssetDatabase.LoadAssetAtPath<GameObject>(path);var importer=AssetImporter.GetAtPath(path) as ModelImporter;
  if(!model||importer==null)throw new InvalidOperationException("Import the hero FBX first: "+path);
  var desc=importer.humanDescription;desc.human=Mapping();
  desc.skeleton=model.GetComponentsInChildren<Transform>(true).Select(t=>new SkeletonBone{name=t.name,position=t.localPosition,rotation=t.localRotation,scale=t.localScale}).ToArray();
  desc.upperArmTwist=.5f;desc.lowerArmTwist=.5f;desc.upperLegTwist=.5f;desc.lowerLegTwist=.5f;desc.armStretch=.05f;desc.legStretch=.05f;
  importer.humanDescription=desc;importer.animationType=ModelImporterAnimationType.Human;importer.avatarSetup=ModelImporterAvatarSetup.CreateFromThisModel;importer.optimizeGameObjects=false;
  if(!AssetDatabase.IsValidFolder("Assets/Editor"))AssetDatabase.CreateFolder("Assets","Editor");
  const string folder="Assets/Editor/CharKitImportMasks";if(!AssetDatabase.IsValidFolder(folder))AssetDatabase.CreateFolder("Assets/Editor","CharKitImportMasks");
  string maskPath=folder+"/"+id+"_Job003.mask";var mask=AssetDatabase.LoadAssetAtPath<AvatarMask>(maskPath);if(!mask){mask=new AvatarMask();AssetDatabase.CreateAsset(mask,maskPath);}
  var ts=model.GetComponentsInChildren<Transform>(true);mask.transformCount=ts.Length;
  for(int i=0;i<ts.Length;i++){mask.SetTransformPath(i,AnimationUtility.CalculateTransformPath(ts[i],model.transform));mask.SetTransformActive(i,true);}
  for(int i=0;i<(int)AvatarMaskBodyPart.LastBodyPart;i++)mask.SetHumanoidBodyPartActive((AvatarMaskBodyPart)i,true);
  EditorUtility.SetDirty(mask);AssetDatabase.SaveAssets();
  var clips=importer.defaultClipAnimations;
  if(clips.Length!=17||Names.Any(n=>!clips.Any(c=>c.name==n)))throw new InvalidOperationException("Expected exact 17 job 003 clips in "+path);
  foreach(var clip in clips){clip.loopTime=Loops.Contains(clip.name);clip.ConfigureClipFromMask(mask);clip.maskType=ClipAnimationMaskType.CopyFromOther;clip.maskSource=mask;}
  importer.clipAnimations=clips;importer.SaveAndReimport();
  var avatar=AssetDatabase.LoadAllAssetsAtPath(path).OfType<Avatar>().FirstOrDefault();if(!avatar||!avatar.isValid||!avatar.isHuman)throw new InvalidOperationException("Invalid job 003 Humanoid avatar");
  Debug.Log("Configured job 003 hero: "+id+" (17 clips, full weapon Transform mask)");
 }
