using UnityEditor;

public static class AllHeroes003Check {
 const string W="C:/Users/demog/Documents/Codex/2026-11-08/se-3/work/allheroes003/";
 const string J="E:/AI接案/SpellDuel-repo/chargen/jobs/spellduel-cloud/003-all-heroes/";
 [Serializable] public class Pair {public string humanName,boneName;}
 [Serializable] public class Request {public Pair[] mapping;public string[] classes;}
 [Serializable] public class ClipInfo {public string name;public float seconds,fps;public bool loop;public int attachmentCurves;}
 [Serializable] public class Pose {public string clip;public float time;public Vector3 root,rightHand,leftHand,weaponR,weaponL;public Vector3 min,max;public float maxHandSkinDistance;}
 [Serializable] public class Report {public string id,error;public bool ok,humanoidValid,humanoidHuman;public int bodyTriangles,outlineTriangles,materialCount,bones,maxInfluences;public float neutralFeetY,neutralHeight,rootTranslationMax;public ClipInfo[] clips;public Pose[] poses;}
 static readonly string[] Names={"idle","walk","run","walk_back","walk_left","walk_right","run_back","run_left","run_right","attack","skill","skill_loop","hit","crouch","jump","die","victory"};
 static readonly string[] Loops={"idle","walk","run","walk_back","walk_left","walk_right","run_back","run_left","run_right","skill_loop","crouch"};
 static bool IsProp(EditorCurveBinding b)=>b.path.EndsWith("/Weapon_R")||b.path.EndsWith("/Weapon_L")||b.path.EndsWith("/Shield")||b.path.EndsWith("/Book")||b.path.EndsWith("/BowNock")||b.path.EndsWith("/Arrow");
 public static void Execute(){bool ok=true;var req=JsonUtility.FromJson<Request>(File.ReadAllText(W+"unity-request.json"));foreach(string id in req.classes){var report=new Report{id=id};try{Check(id,req,report);report.ok=true;}catch(Exception e){report.error=e.ToString();Debug.LogException(e);ok=false;}Directory.CreateDirectory(J+id);File.WriteAllText(J+id+"/unity-inspect.json",JsonUtility.ToJson(report,true));Debug.Log("JOB003_CHECK "+id+" "+report.ok);}EditorApplication.Exit(ok?0:2);}
 static void Check(string id,Request req,Report r){
  AssetDatabase.Refresh();string folder="Assets/AllHeroes003/"+id+"_v1/",path=folder+id+".fbx";var model=AssetDatabase.LoadAssetAtPath<GameObject>(path);var imp=(ModelImporter)AssetImporter.GetAtPath(path);if(model==null||imp==null)throw new Exception("Missing model");
  var desc=imp.humanDescription;desc.human=req.mapping.Select(m=>new HumanBone{humanName=m.humanName,boneName=m.boneName,limit=new HumanLimit{useDefaultValues=true}}).ToArray();desc.skeleton=model.GetComponentsInChildren<Transform>(true).Select(t=>new SkeletonBone{name=t.name,position=t.localPosition,rotation=t.localRotation,scale=t.localScale}).ToArray();desc.upperArmTwist=.5f;desc.lowerArmTwist=.5f;desc.upperLegTwist=.5f;desc.lowerLegTwist=.5f;desc.armStretch=.05f;desc.legStretch=.05f;
  imp.humanDescription=desc;imp.animationType=ModelImporterAnimationType.Human;imp.avatarSetup=ModelImporterAvatarSetup.CreateFromThisModel;imp.optimizeGameObjects=false;
  string maskPath="Assets/AllHeroes003/"+id+"_Full.mask";var mask=AssetDatabase.LoadAssetAtPath<AvatarMask>(maskPath);if(!mask){mask=new AvatarMask();AssetDatabase.CreateAsset(mask,maskPath);}var ts=model.GetComponentsInChildren<Transform>(true);mask.transformCount=ts.Length;for(int i=0;i<ts.Length;i++){mask.SetTransformPath(i,AnimationUtility.CalculateTransformPath(ts[i],model.transform));mask.SetTransformActive(i,true);}for(int i=0;i<(int)AvatarMaskBodyPart.LastBodyPart;i++)mask.SetHumanoidBodyPartActive((AvatarMaskBodyPart)i,true);EditorUtility.SetDirty(mask);AssetDatabase.SaveAssets();
  var settings=imp.defaultClipAnimations;foreach(var c in settings){c.loopTime=Loops.Contains(c.name);c.ConfigureClipFromMask(mask);c.maskType=ClipAnimationMaskType.CopyFromOther;c.maskSource=mask;}imp.clipAnimations=settings;imp.SaveAndReimport();
  var assets=AssetDatabase.LoadAllAssetsAtPath(path);var avatar=assets.OfType<Avatar>().First();r.humanoidValid=avatar.isValid;r.humanoidHuman=avatar.isHuman;if(!avatar.isValid||!avatar.isHuman)throw new Exception("Invalid Humanoid");var clips=assets.OfType<AnimationClip>().Where(c=>!c.name.StartsWith("__preview")).ToArray();if(clips.Length!=17||Names.Any(n=>!clips.Any(c=>c.name==n)))throw new Exception("Expected exact 17 named clips");
  r.clips=clips.Select(c=>new ClipInfo{name=c.name,seconds=c.length,fps=c.frameRate,loop=c.isLooping,attachmentCurves=AnimationUtility.GetCurveBindings(c).Count(IsProp)}).ToArray();foreach(var c in r.clips)if(c.fps!=30||c.loop!=Loops.Contains(c.name)||c.attachmentCurves==0)throw new Exception("Clip import contract "+c.name);
  EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Single);model=AssetDatabase.LoadAssetAtPath<GameObject>(path);var root=(GameObject)PrefabUtility.InstantiatePrefab(model);var anim=root.GetComponent<Animator>();anim.avatar=avatar;anim.applyRootMotion=false;anim.cullingMode=AnimatorCullingMode.AlwaysAnimate;
  var main=new Material(Shader.Find("SpellDuel/CharCel"));main.mainTexture=AssetDatabase.LoadAssetAtPath<Texture2D>(folder+id+"_albedo.png");main.SetTexture("_EmissionMap",AssetDatabase.LoadAssetAtPath<Texture2D>(folder+id+"_emission.png"));var outline=new Material(Shader.Find("SpellDuel/CharOutline"));var skins=root.GetComponentsInChildren<SkinnedMeshRenderer>();var mats=new HashSet<string>();
  r.neutralFeetY=float.MaxValue;float high=float.MinValue;foreach(var skin in skins){bool shell=skin.name.StartsWith("OUTLINE_");var mesh=skin.sharedMesh;if(shell)r.outlineTriangles+=mesh.triangles.Length/3;else{r.bodyTriangles+=mesh.triangles.Length/3;foreach(var v in mesh.vertices){var p=skin.transform.TransformPoint(v);r.neutralFeetY=Math.Min(r.neutralFeetY,p.y);high=Math.Max(high,p.y);}}foreach(var m in skin.sharedMaterials)mats.Add(m.name);skin.sharedMaterials=skin.sharedMaterials.Select(m=>shell?outline:main).ToArray();r.bones=Math.Max(r.bones,skin.bones.Length);foreach(var b in mesh.boneWeights){int count=(b.weight0>0?1:0)+(b.weight1>0?1:0)+(b.weight2>0?1:0)+(b.weight3>0?1:0);if(count==0)throw new Exception("Unweighted vertex in "+skin.name);r.maxInfluences=Math.Max(r.maxInfluences,count);}}
  r.materialCount=mats.Count;r.neutralHeight=high-r.neutralFeetY;if(r.bodyTriangles>15000||r.materialCount>2||r.maxInfluences>4)throw new Exception("Geometry budget");if(Math.Abs(r.neutralFeetY)>.03f)throw new Exception("Feet origin "+r.neutralFeetY);
  var light=new GameObject("Key").AddComponent<Light>();light.type=LightType.Directional;light.transform.rotation=Quaternion.Euler(30,150,0);var cam=new GameObject("Preview camera").AddComponent<Camera>();cam.orthographic=true;cam.orthographicSize=1.30f;cam.backgroundColor=new Color(.24f,.265f,.32f);cam.clearFlags=CameraClearFlags.SolidColor;cam.nearClipPlane=.01f;cam.farClipPlane=30;var rt=new RenderTexture(720,900,24);rt.antiAliasing=4;cam.targetTexture=rt;
  string controllerPath="Assets/AllHeroes003/"+id+"_QA.controller";AssetDatabase.DeleteAsset(controllerPath);var controller=AnimatorController.CreateAnimatorControllerAtPath(controllerPath);foreach(var c in clips){var st=controller.layers[0].stateMachine.AddState(c.name);st.motion=c;}anim.runtimeAnimatorController=controller;
  string prev=J+id+"/previews/";Directory.CreateDirectory(prev);Directory.CreateDirectory(W+id+"/frames");var poses=new List<Pose>();
  foreach(var c in clips)for(int phase=0;phase<3;phase++){
   float time=phase==2?.999999f:phase*.5f;Sample(anim,c.name,time);var snapshots=Bake(skins,root,c.name,time,out var pose);if(pose.maxHandSkinDistance>.30f)throw new Exception("Hand skin detached: "+pose.maxHandSkinDistance+" in "+c.name);poses.Add(pose);r.rootTranslationMax=Math.Max(r.rootTranslationMax,pose.root.magnitude);
   if(phase==1){CameraAt(cam,new Vector3(1.1f,1.25f,4),new Vector3(0,1.0f,0));Render(cam,rt,W+id+"/frames/pose_"+c.name+".png");}
   if(c.name=="idle"&&phase==0){CameraAt(cam,new Vector3(0,1.10f,4),new Vector3(0,1.0f,0));Render(cam,rt,prev+"front.png");CameraAt(cam,new Vector3(0,1.10f,-4),new Vector3(0,1.0f,0));Render(cam,rt,prev+"back.png");cam.orthographic=false;cam.fieldOfView=42;CameraAt(cam,new Vector3(-.55f,2.9f,-3.5f),new Vector3(0,.9f,0));Render(cam,rt,prev+"third_person.png");cam.orthographic=true;}
   Clear(skins,snapshots);
  }
  for(int i=0;i<13;i++){float t=Mathf.Min(.999999f,i/12f);Sample(anim,"attack",t);var snaps=Bake(skins,root,"attack",t,out var pose);CameraAt(cam,new Vector3(1.1f,1.25f,4),new Vector3(0,1.0f,0));Render(cam,rt,W+id+"/frames/attack_"+i.ToString("D2")+".png");if(i==7)Render(cam,rt,prev+"attack.png");Clear(skins,snaps);}
  r.poses=poses.ToArray();if(r.rootTranslationMax>1e-6f)throw new Exception("Root displacement");RenderTexture.active=null;cam.targetTexture=null;rt.Release();
 }
 static void Sample(Animator a,string clip,float t){a.Rebind();a.Update(0);a.Play(clip,0,t);a.Update(0);}
 static Vector3 Find(GameObject root,string name){var t=root.GetComponentsInChildren<Transform>().FirstOrDefault(x=>x.name==name);return t?t.position:Vector3.zero;}
 static List<GameObject> Bake(SkinnedMeshRenderer[] skins,GameObject root,string clip,float time,out Pose pose){pose=new Pose{clip=clip,time=time,root=root.transform.position,rightHand=Find(root,"J_Bip_R_Hand"),leftHand=Find(root,"J_Bip_L_Hand"),weaponR=Find(root,"Weapon_R"),weaponL=Find(root,"Weapon_L"),min=Vector3.one*999,max=Vector3.one*-999};var result=new List<GameObject>();foreach(var skin in skins){var m=new Mesh();skin.BakeMesh(m,false);var go=new GameObject("CPU pose");go.transform.position=skin.transform.position;go.transform.rotation=skin.transform.rotation;go.transform.localScale=skin.transform.lossyScale;go.AddComponent<MeshFilter>().sharedMesh=m;go.AddComponent<MeshRenderer>().sharedMaterials=skin.sharedMaterials;result.Add(go);skin.enabled=false;var verts=m.vertices;var weights=skin.sharedMesh.boneWeights;for(int vi=0;vi<verts.Length;vi++){var p=skin.transform.TransformPoint(verts[vi]);pose.min=Vector3.Min(pose.min,p);pose.max=Vector3.Max(pose.max,p);var b=skin.bones[weights[vi].boneIndex0].name;if(b.Contains("_Hand")||b.Contains("_Thumb")||b.Contains("_Index")||b.Contains("_Middle")||b.Contains("_Ring")||b.Contains("_Little")){var hand=b.Contains("_R_")?pose.rightHand:pose.leftHand;pose.maxHandSkinDistance=Math.Max(pose.maxHandSkinDistance,Vector3.Distance(p,hand));}}}return result;}
 static void Clear(SkinnedMeshRenderer[] skins,List<GameObject> snapshots){foreach(var go in snapshots){UnityEngine.Object.DestroyImmediate(go.GetComponent<MeshFilter>().sharedMesh);UnityEngine.Object.DestroyImmediate(go);}foreach(var s in skins)s.enabled=true;}
 static void CameraAt(Camera c,Vector3 p,Vector3 target){c.transform.position=p;c.transform.LookAt(target);}
 static void Render(Camera c,RenderTexture rt,string path){c.Render();RenderTexture.active=rt;var tex=new Texture2D(rt.width,rt.height,TextureFormat.RGB24,false);tex.ReadPixels(new Rect(0,0,rt.width,rt.height),0,0);tex.Apply();File.WriteAllBytes(path,tex.EncodeToPNG());UnityEngine.Object.DestroyImmediate(tex);}
}
